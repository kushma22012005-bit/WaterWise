required = ["Date", "Time", "Consumption", "Temperature", "Rain", "Sun"]
missing = [c for c in required if c not in df_raw.columns]
if missing:
    raise ValueError(f"Dataset is missing required columns: {missing}")

df = df_raw[required].copy()

raw_date = df["Date"].astype(str).str.strip()
parsed_dmy = pd.to_datetime(raw_date, format="%d-%m-%Y", errors="coerce")
parsed_slash = pd.to_datetime(raw_date, format="%m/%d/%Y", errors="coerce")
df["DateTime"] = parsed_dmy.fillna(parsed_slash)

df["Hour"] = df["Time"].astype(str).str.extract(r"(\d{1,2}):\d{2}", expand=False)
df["Hour"] = pd.to_numeric(df["Hour"], errors="coerce")
df["Consumption"] = pd.to_numeric(df["Consumption"], errors="coerce")
for col in ["Temperature", "Rain", "Sun"]:
    df[col] = pd.to_numeric(df[col], errors="coerce")

rows_before = len(df)
df = df.dropna(subset=["DateTime", "Hour", "Consumption", "Temperature", "Sun"]).copy()
missing_rain_before_fill = int(df["Rain"].isna().sum())
df["Rain"] = df["Rain"].fillna(df["Rain"].median())

df["DateTime"] = df["DateTime"] + pd.to_timedelta(df["Hour"], unit="h")
df = df.sort_values("DateTime").drop_duplicates(subset=["DateTime"]).reset_index(drop=True)

df["Year"] = df["DateTime"].dt.year
df["Month"] = df["DateTime"].dt.month
df["Day"] = df["DateTime"].dt.day
df["DayOfWeek"] = df["DateTime"].dt.dayofweek
df["Hour"] = df["DateTime"].dt.hour
df["DayOfYear"] = df["DateTime"].dt.dayofyear

print("Rows before cleaning:", rows_before)
print("Rows after cleaning:", len(df))
print("Missing Rain values filled:", missing_rain_before_fill)
print("Remaining missing values in modeling columns:")
display(df[["DateTime", "Consumption", "Temperature", "Rain", "Sun"]].isna().sum().to_frame("Missing values"))
display(df.head())

# Conclusion
- The dataset was inspected and cleaned using the same preprocessing rules as `model.py`.
- Potential consumption outliers were visualized and flagged for inspection; they were not automatically removed.
- A Random Forest regression model was trained on the earlier 80% of the time series and tested on the later 20%.
- MAE, RMSE, and R² were used to evaluate predictions.
- A 24-hour forecast was produced using the latest observed weather values.

# Important limitations
1. The forecast assumes the latest weather readings continue into the future.
2. A chronological holdout test is more realistic than a random split, but performance may change over time.
3. Feature importance shows model associations, not proof of causation.
4. The notebook and website should use the same `model.py` settings to keep results consistent.
